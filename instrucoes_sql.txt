-- 1. TABELA DE PERFIS (PROFILES)
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE,
  full_name TEXT,
  avatar_url TEXT,
  reading_goal INTEGER DEFAULT 12,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- 2. TABELA DE LIVROS (BOOKS)
CREATE TABLE IF NOT EXISTS public.books (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  title TEXT NOT NULL,
  author TEXT NOT NULL,
  pages INTEGER DEFAULT 0,
  genre TEXT,
  type TEXT DEFAULT 'Livro',
  status TEXT DEFAULT 'Não lido',
  rating DECIMAL(3,1),
  cover_image_url TEXT,
  summary TEXT,
  notes TEXT,
  estimated_price DECIMAL(10,2),
  buy_link TEXT,
  current_page INTEGER DEFAULT 0,
  date_added DATE DEFAULT CURRENT_DATE NOT NULL,
  date_started DATE,
  date_finished DATE,
  days_to_finish INTEGER,
  times_read INTEGER DEFAULT 1,
  was_wishlist BOOLEAN DEFAULT FALSE
);

-- 3. MIGRAÇÕES CRÍTICAS (Correção do erro 22P02 - Integer vs Decimal)
DO $$ 
BEGIN 
  -- Converte rating para DECIMAL se ainda for INTEGER
  IF EXISTS (
    SELECT 1 FROM information_schema.columns 
    WHERE table_name='books' AND column_name='rating' AND data_type='integer'
  ) THEN
    ALTER TABLE public.books ALTER COLUMN rating TYPE DECIMAL(3,1) USING rating::DECIMAL(3,1);
  END IF;

  -- Garante que current_page existe
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='books' AND column_name='current_page') THEN
    ALTER TABLE public.books ADD COLUMN current_page INTEGER DEFAULT 0;
  END IF;

  -- Garante que was_wishlist existe
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='books' AND column_name='was_wishlist') THEN
    ALTER TABLE public.books ADD COLUMN was_wishlist BOOLEAN DEFAULT FALSE;
  END IF;
  
  -- Garante que times_read existe
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='books' AND column_name='times_read') THEN
    ALTER TABLE public.books ADD COLUMN times_read INTEGER DEFAULT 1;
  END IF;
END $$;

-- 4. GATILHO PARA CRIAÇÃO AUTOMÁTICA DE PERFIL
CREATE OR REPLACE FUNCTION public.handle_new_user() 
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, full_name, avatar_url)
  VALUES (new.id, new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'avatar_url')
  ON CONFLICT (id) DO NOTHING;
  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- 5. HABILITAR RLS
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.books ENABLE ROW LEVEL SECURITY;

-- 6. POLÍTICAS DE SEGURANÇA
-- (Repetir políticas existentes para garantir que estão ativas)
DROP POLICY IF EXISTS "Usuários podem ver seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem ver seus próprios livros" ON public.books FOR SELECT USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem inserir seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem inserir seus próprios livros" ON public.books FOR INSERT WITH CHECK (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem atualizar seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem atualizar seus próprios livros" ON public.books FOR UPDATE USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem deletar seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem deletar seus próprios livros" ON public.books FOR DELETE USING (auth.uid() = user_id);

-- 7. PERMISSÕES
GRANT ALL ON TABLE public.books TO authenticated;
GRANT ALL ON TABLE public.profiles TO authenticated;
