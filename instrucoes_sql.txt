-- 1. TABELA DE PERFIS (PROFILES)
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE,
  full_name TEXT,
  avatar_url TEXT,
  reading_goal INTEGER DEFAULT 12,
  gemini_api_key TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- 2. TABELA DE LIVROS (BOOKS)
CREATE TABLE IF NOT EXISTS public.books (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  title TEXT NOT NULL,
  author TEXT NOT NULL,
  pages INTEGER DEFAULT 0,
  genre TEXT,
  type TEXT DEFAULT 'Livro',
  status TEXT DEFAULT 'Não lido',
  rating DECIMAL(3,1),
  cover_image_url TEXT,
  summary TEXT,
  notes TEXT,
  estimated_price DECIMAL(10,2),
  buy_link TEXT,
  current_page INTEGER DEFAULT 0,
  date_added DATE DEFAULT CURRENT_DATE NOT NULL,
  date_started DATE,
  date_finished DATE,
  days_to_finish INTEGER,
  times_read INTEGER DEFAULT 1,
  was_wishlist BOOLEAN DEFAULT FALSE
);

-- 2.1 MIGRAÇÃO: ADICIONAR COLUNAS CASO NÃO EXISTAM (Para quem já tem a tabela antiga)
DO $$ 
BEGIN 
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='books' AND column_name='was_wishlist') THEN
    ALTER TABLE public.books ADD COLUMN was_wishlist BOOLEAN DEFAULT FALSE;
  END IF;
END $$;

DO $$ 
BEGIN 
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='books' AND column_name='times_read') THEN
    ALTER TABLE public.books ADD COLUMN times_read INTEGER DEFAULT 1;
  END IF;
END $$;

-- 3. HABILITAR RLS (Row Level Security)
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.books ENABLE ROW LEVEL SECURITY;

-- 4. POLÍTICAS PARA A TABELA 'BOOKS' (REMOVER E CRIAR NOVAMENTE)
DROP POLICY IF EXISTS "Usuários podem ver seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem ver seus próprios livros" 
ON public.books FOR SELECT 
USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem inserir seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem inserir seus próprios livros" 
ON public.books FOR INSERT 
WITH CHECK (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem atualizar seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem atualizar seus próprios livros" 
ON public.books FOR UPDATE 
USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Usuários podem deletar seus próprios livros" ON public.books;
CREATE POLICY "Usuários podem deletar seus próprios livros" 
ON public.books FOR DELETE 
USING (auth.uid() = user_id);

-- 5. POLÍTICAS PARA A TABELA 'PROFILES' (REMOVER E CRIAR NOVAMENTE)
DROP POLICY IF EXISTS "Usuários podem ver seu próprio perfil" ON public.profiles;
CREATE POLICY "Usuários podem ver seu próprio perfil" 
ON public.profiles FOR SELECT 
USING (auth.uid() = id);

DROP POLICY IF EXISTS "Usuários podem atualizar seu próprio perfil" ON public.profiles;
CREATE POLICY "Usuários podem atualizar seu próprio perfil" 
ON public.profiles FOR UPDATE 
USING (auth.uid() = id);

DROP POLICY IF EXISTS "Usuários podem inserir seu próprio perfil" ON public.profiles;
CREATE POLICY "Usuários podem inserir seu próprio perfil" 
ON public.profiles FOR INSERT 
WITH CHECK (auth.uid() = id);

-- 6. GARANTIR QUE O SCHEMA PUBLIC TEM PERMISSÕES
GRANT ALL ON TABLE public.books TO authenticated;
GRANT ALL ON TABLE public.profiles TO authenticated;
GRANT ALL ON TABLE public.books TO service_role;
GRANT ALL ON TABLE public.profiles TO service_role;

-- 7. FORÇAR ATUALIZAÇÃO DO CACHE DO API
NOTIFY pgrst, 'reload schema';
