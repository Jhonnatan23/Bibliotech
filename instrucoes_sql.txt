
-- ======================================================
-- SCRIPT DE LIMPEZA E OTIMIZAÇÃO: BIBLIOTECH (v18)
-- REMOVE COLUNAS DE IMAGEM E CONFIGURAÇÕES DE STORAGE
-- ======================================================

-- 1. REMOÇÃO DA COLUNA DE IMAGEM (Limpeza de Dados)
ALTER TABLE public.books DROP COLUMN IF EXISTS cover_image_url;

-- 2. LIMPEZA DE STORAGE (Opcional: Remove acesso ao bucket)
-- Nota: Para deletar os arquivos físicos, use o painel Storage do Supabase e delete o bucket 'book-covers'
DROP POLICY IF EXISTS "Capas são públicas" ON storage.objects;
DROP POLICY IF EXISTS "Usuários gerenciam suas próprias capas" ON storage.objects;
DROP POLICY IF EXISTS "Usuários atualizam suas próprias capas" ON storage.objects;
DROP POLICY IF EXISTS "Usuários deletam suas próprias capas" ON storage.objects;

-- 3. GARANTIA DE ESTRUTURA PARA PERFIS E LIVROS
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE,
  full_name TEXT,
  avatar_url TEXT,
  reading_goal INTEGER DEFAULT 12,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS public.books (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
  title TEXT NOT NULL,
  author TEXT NOT NULL,
  pages INTEGER DEFAULT 0,
  genre TEXT,
  type TEXT DEFAULT 'Livro',
  status TEXT DEFAULT 'Não lido',
  rating DECIMAL(3,1),
  summary TEXT,
  notes TEXT,
  estimated_price DECIMAL(10,2),
  price_paid DECIMAL(10,2),
  buy_link TEXT,
  current_page INTEGER DEFAULT 0,
  date_added DATE DEFAULT CURRENT_DATE NOT NULL,
  date_started DATE,
  date_finished DATE,
  days_to_finish INTEGER,
  times_read INTEGER DEFAULT 1,
  was_wishlist BOOLEAN DEFAULT FALSE
);

-- 4. RE-SINCERIZAÇÃO DE ÍNDICES
CREATE INDEX IF NOT EXISTS idx_books_user_id ON public.books(user_id);
CREATE INDEX IF NOT EXISTS idx_books_status ON public.books(status);
CREATE INDEX IF NOT EXISTS idx_books_date_added ON public.books(date_added DESC);
